    %% ===== 1. TÀI KHOẢN & PHÂN QUYỀN =====

    ROLE {
        integer role_id PK "NOT NULL"
        text role_name UK "NOT NULL; USER, DOCTOR, RECEPTIONIST, NURSE, ADMIN"
        text description
        integer requires_approval "NOT NULL; 0 hoặc 1"
    }

    USERS {
        integer user_id PK "NOT NULL"
        integer role_id FK "NOT NULL"
        text full_name "NOT NULL"
        text phone UK
        text email UK
        text password_hash "NOT NULL; hash mật khẩu, không rỗng"
        text approval_status "NOT NULL; PENDING, APPROVED, REJECTED"
        text account_status "NOT NULL; ACTIVE, LOCKED"
        integer email_verified "NOT NULL; 0 hoặc 1"
        timestamptz deleted_at
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    AUTH_TOKEN {
        integer token_id PK "NOT NULL"
        integer user_id FK "NOT NULL"
        text token_type "NOT NULL; RESET_PASSWORD, VERIFY_EMAIL, OTP"
        text token_hash "NOT NULL"
        timestamptz expires_at "NOT NULL"
        timestamptz used_at
        timestamptz created_at "NOT NULL"
    }

    %% ===== 2. HỒ SƠ BỆNH NHÂN (tách khỏi tài khoản) =====

    PATIENT {
        integer patient_id PK "NOT NULL"
        integer user_id FK "chủ tài khoản, NULL nếu walk-in chưa có TK"
        text full_name "NOT NULL"
        date dob
        text gender
        text id_card
        text address
        text phone
        timestamptz archived_at "NULL khi còn sử dụng"
        text health_insurance
        text relationship "SELF, PARENT, CHILD, SPOUSE, OTHER"
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    %% ===== 3. BÁC SĨ, KHOA, PHÒNG =====

    DEPARTMENT {
        integer department_id PK "NOT NULL"
        text name "NOT NULL"
        text description
        integer is_active "NOT NULL; 0 hoặc 1"
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    DOCTOR_PROFILE {
        integer doctor_profile_id PK "NOT NULL"
        integer user_id FK "NOT NULL; UNIQUE"
        integer department_id FK "NOT NULL"
        text academic_degree
        text specialization
        integer experience_years "NOT NULL"
        text introduction
        bigint consultation_fee "NOT NULL"
        text avatar_url
        integer is_active "NOT NULL; 0 hoặc 1"
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    ROOM {
        integer room_id PK "NOT NULL"
        integer department_id FK "NOT NULL"
        text room_name "NOT NULL"
        text description
        integer is_active "NOT NULL; 0 hoặc 1"
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    %% ===== 4. LỊCH LÀM VIỆC =====
    %% UNIQUE (doctor_profile_id, work_date, start_time)
    %% UNIQUE (room_id, work_date, start_time)

    WORK_SCHEDULE {
        integer schedule_id PK "NOT NULL"
        integer doctor_profile_id FK "NOT NULL"
        integer room_id FK "NOT NULL"
        date work_date "NOT NULL"
        time start_time "NOT NULL"
        time end_time "NOT NULL"
        integer max_quota "NOT NULL"
        integer booked_count "NOT NULL; cập nhật trong transaction"
        text schedule_status "NOT NULL; OPEN, FULL, CLOSED, UNAVAILABLE"
        text unavailable_reason
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    %% ===== 5. LƯỢT KHÁM (gộp online + walk-in) =====
    %% UNIQUE (schedule_id, queue_number)
    %% department suy ra từ doctor_profile

    ENCOUNTER {
        integer encounter_id PK "NOT NULL"
        text encounter_type "NOT NULL; ONLINE, WALK_IN"
        integer patient_id FK "NOT NULL"
        integer doctor_profile_id FK "NOT NULL"
        integer schedule_id FK "NOT NULL; online và walk-in đều có ca"
        integer room_id FK
        integer created_by_id FK "người tạo lượt khám, có thể NULL"
        bigint consultation_fee_snapshot "VND; NULL với dữ liệu cũ chưa rõ giá"
        bigint deposit_amount_snapshot "VND; NULL với dữ liệu cũ chưa rõ giá"
        text symptoms
        text encounter_status "NOT NULL; HOLDING, CONFIRMED, CHECKED_IN, IN_PROGRESS, COMPLETED, CANCELLED, NO_SHOW, EXPIRED"
        integer queue_number
        timestamptz estimated_exam_at
        text qr_code
        timestamptz hold_expires_at "chỉ online"
        timestamptz checked_in_at
        integer cancelled_by_id FK
        text cancel_reason
        timestamptz cancelled_at
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

    ENCOUNTER_STATUS_LOG {
        integer log_id PK "NOT NULL"
        integer encounter_id FK "NOT NULL"
        text old_status
        text new_status "NOT NULL"
        integer changed_by_id FK "NOT NULL"
        text note
        timestamptz changed_at "NOT NULL"
    }

    ENCOUNTER_TRANSFER_LOG {
        integer log_id PK "NOT NULL"
        integer encounter_id FK "NOT NULL"
        integer old_doctor_id FK "NOT NULL"
        integer new_doctor_id FK "NOT NULL"
        integer transferred_by_id FK "NOT NULL"
        text reason
        timestamptz transferred_at "NOT NULL"
    }

    %% ===== 6. THANH TOÁN =====

    PAYMENT {
        integer payment_id PK "NOT NULL"
        integer encounter_id FK "NOT NULL"
        text payment_type "NOT NULL; DEPOSIT, EXAM_FEE, REFUND"
        bigint amount "NOT NULL"
        text payment_method "NOT NULL; CASH, BANK, EWALLET, CARD"
        text transaction_status "NOT NULL; PENDING, SUCCESS, FAILED, REFUNDED"
        text transaction_code UK
        timestamptz paid_at
        timestamptz refunded_at
        timestamptz created_at "NOT NULL"
        timestamptz updated_at "NOT NULL"
    }

        timestamptz updated_at "NOT NULL"
    }

    ROLE_REQUEST {
        integer request_id PK "NOT NULL"
        integer user_id FK "NOT NULL"
        integer requested_role_id FK "NOT NULL"

        timestamptz updated_at "NOT NULL"
    }
